```swift
import SwiftUI

struct StatCard<Content: View>: View {
    let title: String
    let value: String
    let icon: String
    let theme: any Theme
    let content: Content

    init(title: String, value: String, icon: String, theme: any Theme, @ViewBuilder content: () -> Content) {
        self.title = title
        self.value = value
        self.icon = icon
        self.theme = theme
        self.content = content()
    }

    var body: some View {
        VStack(alignment: .leading, spacing: 8) {
            HStack {
                Image(systemName: icon)
                    .foregroundColor(theme.primaryColor)
                Text(title)
                    .font(.caption)
                    .foregroundColor(theme.secondaryTextColor)
            }
            Text(value)
                .font(.title2.bold())
                .foregroundColor(theme.primaryTextColor)
            content
        }
        .padding()
        .background(theme.cardBackgroundColor)
        .cornerRadius(theme.cardCornerRadius)
    }
}
```

**Step 2: 创建 DashboardViewModel.swift**

```swift
import SwiftUI
import Combine

@MainActor
class DashboardViewModel: ObservableObject {
    @Published var cpuUsage: Double = 0
    @Published var memoryPercent: Double = 0
    @Published var diskPercent: Double = 0
    @Published var networkSent: UInt64 = 0
    @Published var networkRecv: UInt64 = 0

    private var cancellables = Set<AnyCancellable>()

    init() {
        MonitorService.shared.$stats
            .compactMap { $0 }
            .receive(on: DispatchQueue.main)
            .sink { [weak self] stats in
                self?.updateFromStats(stats)
            }
            .store(in: &cancellables)
    }

    private func updateFromStats(_ stats: SystemStats) {
        cpuUsage = stats.cpu.usagePercent
        memoryPercent = stats.memory.usedPercent
        diskPercent = stats.disk.first?.usedPercent ?? 0
        networkSent = stats.network.reduce(0) { $0 + $1.bytesSent }
        networkRecv = stats.network.reduce(0) { $0 + $1.bytesRecv }
    }
}
```

**Step 3: 创建 DashboardView.swift**

```swift
import SwiftUI
import Charts

struct DashboardView: View {
    @StateObject private var viewModel = DashboardViewModel()
    @ObservedObject var themeManager: ThemeManager

    var body: some View {
        let theme = themeManager.currentTheme

        ScrollView {
            VStack(spacing: 16) {
                // 4个核心指标卡片
                LazyVGrid(columns: [GridItem(.flexible()), GridItem(.flexible())], spacing: 12) {
                    StatCard(title: "CPU", value: String(format: "%.1f%%", viewModel.cpuUsage), icon: "cpu", theme: theme) {
                        ProgressView(value: viewModel.cpuUsage / 100)
                            .tint(theme.primaryColor)
                    }

                    StatCard(title: "内存", value: String(format: "%.1f%%", viewModel.memoryPercent), icon: "memorychip", theme: theme) {
                        ProgressView(value: viewModel.memoryPercent / 100)
                            .tint(theme.secondaryColor)
                    }

                    StatCard(title: "磁盘", value: String(format: "%.1f%%", viewModel.diskPercent), icon: "internaldrive", theme: theme) {
                        ProgressView(value: viewModel.diskPercent / 100)
                            .tint(theme.accentColor)
                    }

                    StatCard(title: "网络", value: "\(formatBytes(viewModel.networkRecv))/s", icon: "network", theme: theme) {
                        HStack {
                            Image(systemName: "arrow.down")
                            Text(formatBytes(viewModel.networkRecv))
                        }
                        .font(.caption)
                        .foregroundColor(theme.secondaryTextColor)
                    }
                }

                // 快速入口
                HStack(spacing: 12) {
                    NavigationLink(destination: ProcessListView(themeManager: themeManager)) {
                        Label("进程", systemImage: "list.bullet")
                    }
                    .buttonStyle(themeCardButtonStyle(theme: theme))

                    NavigationLink(destination: AppListView(themeManager: themeManager)) {
                        Label("应用", systemImage: "app.badge")
                    }
                    .buttonStyle(themeCardButtonStyle(theme: theme))
                }
            }
            .padding()
        }
        .background(theme.backgroundColor)
        .navigationTitle("仪表盘")
    }

    private func formatBytes(_ bytes: UInt64) -> String {
        let formatter = ByteCountFormatter()
        formatter.countStyle = .binary
        return formatter.string(fromByteCount: Int64(bytes))
    }

    private func themeCardButtonStyle(theme: any Theme) -> some ButtonStyle {
        PlainButtonStyle {
            Text("测试")
                .foregroundColor(theme.primaryTextColor)
        }
    }
}
```

```swift
import SwiftUI
import Combine

@MainActor
class ProcessListViewModel: ObservableObject {
    @Published var processes: [ProcessInfo] = []
    @Published var sortOrder: SortOrder = .cpu

    enum SortOrder {
        case cpu, memory
    }

    var sortedProcesses: [ProcessInfo] {
        switch sortOrder {
        case .cpu:
            return processes.sorted { $0.cpuPercent > $1.cpuPercent }
        case .memory:
            return processes.sorted { $0.memoryBytes > $1.memoryBytes }
        }
    }

    private var cancellables = Set<AnyCancellable>()

    init() {
        MonitorService.shared.$stats
            .compactMap { $0 }
            .receive(on: DispatchQueue.main)
            .sink { [weak self] stats in
                self?.processes = Array(stats.processes.prefix(50))
            }
            .store(in: &cancellables)
    }
}
```

**Step 2: 创建 ProcessListView.swift**

```swift
    var body: some View {
        let theme = themeManager.currentTheme

        List {
            // 主题设置
            Section {
                ForEach(themeManager.availableThemes, id: \.id) { themeItem in
                    Button(action: {
                        themeManager.setTheme(themeItem)
                    }) {
                        HStack {
                            Circle()
                                .fill(themeItem.primaryColor)
                                .frame(width: 24, height: 24)
                            Text(themeItem.name)
                                .foregroundColor(themeManager.themeId == themeItem.id ? theme.primaryColor : theme.primaryTextColor)
                            Spacer()
                            if themeManager.themeId == themeItem.id {
                                Image(systemName: "checkmark")
                                    .foregroundColor(theme.primaryColor)
                            }
                        }
                    }
                }
            } header: {
                Text("主题")
                    .foregroundColor(theme.secondaryTextColor)
            }

            // 服务器设置
            Section {
                HStack {
                    Text("服务器地址")
                        .foregroundColor(theme.primaryTextColor)
                    Spacer()
                    TextField("http://192.168.1.x:8080", text: $serverAddress)
                        .textFieldStyle(.roundedBorder)
                        .keyboardType(.URL)
                        .autocapitalization(.none)
                        .onChange(of: serverAddress) { _, newValue in
                            monitorService.updateServerAddress(newValue)
                        }
                }

                VStack(alignment: .leading) {
                    Text("刷新间隔: \(Int(refreshInterval))秒")
                        .foregroundColor(theme.primaryTextColor)
                    Slider(value: $refreshInterval, in: 1...10, step: 1)
                        .onChange(of: refreshInterval) { _, newValue in
                            monitorService.updateRefreshInterval(newValue)
                        }
                }
            } header: {
                Text("连接")
                    .foregroundColor(theme.secondaryTextColor)
            }

            // 连接状态
            Section {
                HStack {
                    Text("连接状态")
                        .foregroundColor(theme.primaryTextColor)
                    Spacer()
                    Circle()
                        .fill(monitorService.isConnected ? Color.green : Color.red)
                        .frame(width: 10, height: 10)
                    Text(monitorService.isConnected ? "已连接" : "未连接")
                        .font(.caption)
                        .foregroundColor(theme.secondaryTextColor)
                }

                if let error = monitorService.errorMessage {
                    Text("错误: \(error)")
                        .font(.caption)
                        .foregroundColor(.red)
                }
            } header: {
                Text("状态")
                    .foregroundColor(theme.secondaryTextColor)
            }
        }
        .listStyle(.insetGrouped)
        .scrollContentBackground(.hidden)
        .background(theme.backgroundColor)
        .navigationTitle("设置")
        .onAppear {
            serverAddress = monitorService.serverAddress
            refreshInterval = 2.0
        }
    }
}
```
